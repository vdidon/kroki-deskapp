\section{Model}\label{sec:model}
We study the displacement $x(t)$ of a damped oscillator,
\begin{equation}\label{eq:ode}
  \ddot x + 2\zeta\omega_0 \dot x + \omega_0^2 x = 0 ,
\end{equation}
whose underdamped solution ($\zeta < 1$) is given in Theorem~\ref{thm:sol}.

\begin{theorem}\label{thm:sol}
For $\zeta<1$ the solution of \eqref{eq:ode} is
\begin{equation}\label{eq:sol}
  x(t) = A\, e^{-\zeta\omega_0 t}\cos\!\big(\omega_d t + \varphi\big),
  \qquad \omega_d = \omega_0\sqrt{1-\zeta^2}.
\end{equation}
\end{theorem}

\begin{lemma}\label{lem:energy}
The energy $E(t)=\tfrac12\dot x^2+\tfrac12\omega_0^2x^2$ decays:
$\dot E = -2\zeta\omega_0\,\dot x^2 \le 0$.
\end{lemma}

\section{Noise}\label{sec:noise}
With a random amplitude $A$ of mean $\E[A]=a$ and variance $\Var(A)=\sigma^2$,
\eqref{eq:sol} gives $\E[x(t)] = a\,e^{-\zeta\omega_0 t}\cos(\omega_d t+\varphi)$.
Lemma~\ref{lem:energy} then bounds the energy of every trajectory
(see Section~\ref{sec:model}).
